☁️ ซิงก์ไป Google Sheets (ไม่ระบุตัวตน)

ส่งเฉพาะข้อมูลที่ไม่ระบุตัวตน: รหัสแฮชของผู้รับบริการ ช่วงอายุ เพศ วิธีใช้ อาการ ความรุนแรง ผลลัพธ์ — ไม่ส่งชื่อ โรคประจำตัว หรือหมายเหตุ ข้อความอิสระ

วิธีตั้งค่า Apps Script (คลิกเพื่อดู)
  1. สร้าง Google Sheet ใหม่ → เมนู ส่วนขยาย → Apps Script
  2. วางโค้ดด้านล่างแทนของเดิม แล้วกด Deploy → New deployment → ประเภท Web app
  3. ตั้ง Execute as: Me และ Who has access: Anyone แล้วคัดลอก URL มาวางด้านบน
  4. กด "สุ่มรหัส" ในแอป แล้วคัดลอกรหัสไปแทนที่ใน TOKEN ของสคริปต์ (ต้องตรงกัน) จากนั้น Deploy ใหม่ — แอปไม่เห็นคำตอบจากสคริปต์ จึงต้องเปิดดูใน Sheet ว่าข้อมูลเข้าจริง
var TOKEN = 'วางรหัสลับที่สุ่มจากแอปตรงนี้';
var KINDS = ['record', 'screen', 'test'];

function out(t) { return ContentService.createTextOutput(t); }

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    if (!e.postData || e.postData.contents.length > 20000) return out('bad');
    var d = JSON.parse(e.postData.contents);
    if (d.token !== TOKEN) return out('denied');
    if (KINDS.indexOf(d.kind) === -1) return out('bad');
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sh = ss.getSheetByName(d.kind) || ss.insertSheet(d.kind);
    if (sh.getLastRow() === 0) sh.appendRow(['time', 'facility']);
    var head = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0];
    Object.keys(d.data).forEach(function (k) {
      if (head.indexOf(k) === -1) {
        head.push(k);
        sh.getRange(1, head.length).setValue(k);
      }
    });
    var row = head.map(function (h) {
      if (h === 'time') return new Date(d.ts);
      if (h === 'facility') return d.facility;
      var v = d.data[h];
      if (v === undefined) return '';
      return Array.isArray(v) ? v.join('; ') : v;
    });
    sh.appendRow(row);
    return out('ok');
  } finally {
    lock.releaseLock();
  }
}
